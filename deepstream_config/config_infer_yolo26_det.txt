# nvinfer configuration for the fetch game's detector.
#
# A plain YOLO detector -- not a pose model -- so that people and tennis balls
# can be found in the same pass. `mecanumbot_onboard_cam_detect_objects` filters
# the 80 COCO classes down to the two it publishes (0 person, 32 sports ball);
# nvinfer is left to report everything, because per-class thresholds here and
# gating in the node would be two places to tune the same decision.
#
# onnx-file, model-engine-file, infer-dims and labelfile-path are REWRITTEN at
# startup by that node, which renders a copy of this file from its model_params
# (imgsz, model_name, precision, models_dir) and hands nvinfer the copy. The
# values below are the ones used when this file is fed to nvinfer directly --
# deepstream-app, gst-launch, a standalone test -- and are relative to this file
# so that they resolve wherever the package is installed. Set
# model_params.nvinfer_config to make the node use this file untouched.
[property]
gpu-id=0
net-scale-factor=0.0039215697906911373
model-color-format=0
onnx-file=../models/imgsz_640/yolo26m.onnx
# DeepStream builds and saves the TensorRT engine here if it does not exist.
model-engine-file=../models/imgsz_640/yolo26m.onnx_b1_gpu0_fp16.engine
labelfile-path=labels_coco.txt
batch-size=1
# 0=FP32, 1=INT8, 2=FP16. Must match the precision in model-engine-file's name,
# otherwise nvinfer saves the engine it builds under a different name than the
# one it looks for and rebuilds from scratch on every launch.
network-mode=2
num-detected-classes=80
# The imgsz the ONNX was exported at (channels;height;width). The engine
# filename does not encode the input size, so an engine built from an earlier
# export is loaded silently and keeps running at its own size; this line is what
# a run can be cross-checked against. The node keeps it in step with the model
# folder it selects.
infer-dims=3;640;640
interval=0
# Must differ from the pose detector's, so the two can run at once.
gie-unique-id=2
process-mode=1
# 0 = detector. The pose config sets 3 (instance segmentation) because that is
# how the pose parser smuggles keypoints out in the mask; nothing here needs a
# mask.
network-type=0
# 4 = no clustering, for a model that does its own NMS end-to-end. This is the
# yolo26 setting from DeepStream-Yolo; an older YOLO wants 2 plus an
# nms-iou-threshold below.
cluster-mode=4
# Letterboxed rather than stretched, and this one is NOT a free choice: a
# tennis ball is recognised by being round, and stretching a 16:9 frame into a
# square input turns every ball into an ellipse of aspect ~1.78. That breaks
# both the shape gate in the node and the range estimate downstream, which reads
# the ball's diameter off the box. The pose config stretches because a person's
# shape is not what identifies them.
maintain-aspect-ratio=1
symmetric-padding=1
#workspace-size=2000

# --- THE CRITICAL PARSER SETTINGS ---
# DeepStream-Yolo's ordinary bounding-box parser, NOT the pose one. Where it
# was built differs per machine, so the node expands ~ and $VARS in this line
# and, if nothing is there, looks for DeepStream-Yolo/ under ~/deepstream_source
# (the robot) and ~/Documents/installed_external (a laptop), writing whichever
# exists into the rendered copy. model_params.custom_lib_path replaces the
# search with one path. nvinfer itself expands nothing: to feed this file to
# deepstream-app directly, write an absolute path here.
parse-bbox-func-name=NvDsInferParseYolo
custom-lib-path=~/deepstream_source/DeepStream-Yolo/nvdsinfer_custom_impl_Yolo/libnvdsinfer_custom_impl_Yolo.so
engine-create-func-name=NvDsInferYoloCudaEngineGet

[class-attrs-all]
# A hard floor: nothing below this ever reaches the node's own gates, whose
# acquire thresholds (person 0.5, ball 0.4) sit above it. Kept low here so that
# the node's hysteresis has something to retain a fading detection with.
pre-cluster-threshold=0.2
topk=300
